import json
from datetime import date
from core.models import Day

class ProductionCalendar:
    """
    Производственный календарь.
    Поддерживает загрузку из JSON, XML (Россия) или ручное заполнение.
    """

    def __init__(self):
        self.days: dict[date, Day] = {}

    def load_from_json(self, path: str):
        """Загрузка из JSON: {'2026-10-01': {'weekend': False, 'holiday': False}}"""
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)

        for iso_date, info in data.items():
            d = date.fromisoformat(iso_date)
            self.days[d] = Day(
                date=d,
                is_weekend=info.get("weekend", False),
                is_holiday=info.get("holiday", False),
                is_short=info.get("short", False),
            )

    def load_from_dict(self, data: dict):
        """Ручное заполнение из словаря."""
        for iso_date, info in data.items():
            d = date.fromisoformat(iso_date)
            self.days[d] = Day(
                date=d,
                is_weekend=info.get("weekend", False),
                is_holiday=info.get("holiday", False),
                is_short=info.get("short", False),
            )

    def is_holiday(self, d: date) -> bool:
        return self.days.get(d, Day(d)).is_holiday

    def is_weekend(self, d: date) -> bool:
        return self.days.get(d, Day(d)).is_weekend

    def is_short(self, d: date) -> bool:
        return self.days.get(d, Day(d)).is_short

    def get_norm_hours(self, year: int, month: int) -> float:
        """Норма часов за месяц (упрощённо: 40 ч/нед)."""
        import calendar
        _, days_in_month = calendar.monthrange(year, month)
        norm = 0.0
        for day_num in range(1, days_in_month + 1):
            d = date(year, month, day_num)
            if self.is_weekend(d) or self.is_holiday(d):
                continue
            norm += 8.0
            if self.is_short(d):
                norm -= 1.0
        return norm

import calendar
from datetime import date, timedelta
from core.models import Engineer, ShiftType, Shift
from core.production_calendar import ProductionCalendar

class ScheduleGenerator:
    """
    Алгоритм распределения смен:
    - ночные (будни) 19:00–10:00 = 15 ч
    - суточные (выходные/праздники) 10:00–10:00 = 24 ч
    - равномерная нагрузка
    """

    NIGHT_HOURS = 15.0
    DAILY_HOURS = 24.0

    def __init__(self, year: int, month: int, engineers: list[str],
                 calendar: ProductionCalendar = None,
                 night_hours: float = 15.0,
                 daily_hours: float = 24.0):
        self.year = year
        self.month = month
        self.calendar = calendar or ProductionCalendar()
        self.night_hours = night_hours
        self.daily_hours = daily_hours

        self.engineers = [Engineer(name=n) for n in engineers]
        _, self.days_in_month = calendar.monthrange(year, month)

    def generate(self) -> list[Engineer]:
        """Основной метод распределения."""
        # 1. Собираем все дни месяца
        days = [date(self.year, self.month, d) for d in range(1, self.days_in_month + 1)]

        # 2. Разделяем на будни и выходные/праздники
        workdays, holidays = [], []
        for d in days:
            if self.calendar.is_weekend(d) or self.calendar.is_holiday(d):
                holidays.append(d)
            else:
                workdays.append(d)

        # 3. Считаем общую нагрузку
        total_hours = len(workdays) * self.night_hours + len(holidays) * self.daily_hours
        target_per_engineer = total_hours / len(self.engineers)

        # 4. Инициализируем расписание
        for eng in self.engineers:
            eng.shifts = [ShiftType.OFF] * self.days_in_month

        # 5. Распределяем ночные смены по кругу
        idx = 0
        for d in workdays:
            eng = self.engineers[idx % len(self.engineers)]
            day_idx = d.day - 1
            eng.shifts[day_idx] = ShiftType.NIGHT
            eng.night_count += 1
            eng.total_hours += self.night_hours
            idx += 1

        # 6. Распределяем суточные смены (выходные/праздники)
        #    с учётом уже набранной нагрузки — берём наименее загруженного
        for d in holidays:
            eng = min(self.engineers, key=lambda e: e.total_hours)
            day_idx = d.day - 1
            if self.calendar.is_holiday(d):
                eng.shifts[day_idx] = ShiftType.HOLIDAY
                eng.holiday_count += 1
            else:
                eng.shifts[day_idx] = ShiftType.DAILY
                eng.daily_count += 1
            eng.total_hours += self.daily_hours

        # 7. Балансировка: если разброс > 1 смены, перекидываем
        self._balance()

        return self.engineers

    def _balance(self):
        """Выравнивание нагрузки между инженерами."""
        max_iter = 100
        for _ in range(max_iter):
            self.engineers.sort(key=lambda e: e.total_hours)
            low, high = self.engineers[0], self.engineers[-1]
            diff = high.total_hours - low.total_hours

            # Если разница меньше половины смены — стоп
            if diff < self.night_hours / 2:
                break

            # Ищем у "high" ночную смену, которую можно отдать "low"
            moved = False
            for i, shift in enumerate(high.shifts):
                if shift == ShiftType.NIGHT:
                    # проверяем, что у low этот день свободен
                    if low.shifts[i] == ShiftType.OFF:
                        low.shifts[i] = ShiftType.NIGHT
                        low.night_count += 1
                        low.total_hours += self.night_hours

                        high.shifts[i] = ShiftType.OFF
                        high.night_count -= 1
                        high.total_hours -= self.night_hours
                        moved = True
                        break
            if not moved:
                break

class StatsWidget(QWidget):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.layout = QVBoxLayout(self)

        self.title = QLabel("📊 Статистика за месяц")
        self.title.setFont(QFont("Arial", 12, QFont.Bold))
        self.layout.addWidget(self.title)

        self.table = QTableWidget()
        self.table.setColumnCount(5)
        self.table.setHorizontalHeaderLabels(
            ["Инженер", "Ночных", "Суточных", "Праздничных", "Часов"]
        )
        self.table.horizontalHeader().setSectionResizeMode(QHeaderView.Stretch)
        self.table.setEditTriggers(QTableWidget.NoEditTriggers)
        self.layout.addWidget(self.table)

        self.summary = QLabel("")
        self.summary.setFont(QFont("Arial", 10))
        self.layout.addWidget(self.summary)

    def update_stats(self, engineers, norm_hours: float):
        self.table.setRowCount(len(engineers))
        total = 0.0

        font_bold = QFont()
        font_bold.setBold(True)

        for r, eng in enumerate(engineers):
            self.table.setItem(r, 0, QTableWidgetItem(eng.name))
            self.table.setItem(r, 1, QTableWidgetItem(str(eng.night_count)))
            self.table.setItem(r, 2, QTableWidgetItem(str(eng.daily_count)))
            self.table.setItem(r, 3, QTableWidgetItem(str(eng.holiday_count)))

            hours_item = QTableWidgetItem(f"{eng.total_hours:.0f}")
            hours_item.setFont(font_bold)

            # Подсветка отклонения
            diff = eng.total_hours - norm_hours
            if abs(diff) <= 2:
                hours_item.setBackground(QBrush(QColor("#C6EFCE")))  # зелёный
            elif diff > 2:
                hours_item.setBackground(QBrush(QColor("#FFC7CE")))  # красный
            else:
                hours_item.setBackground(QBrush(QColor("#FFEB9C")))  # жёлтый

            self.table.setItem(r, 4, hours_item)
            total += eng.total_hours

        # Итоговая строка
        self.summary.setText(
            f"Норма месяца: <b>{norm_hours:.0f} ч</b>   |   "
            f"Итого: <b>{total:.0f} ч</b>   |   "
            f"Отклонение от нормы: <b>{total - norm_hours * len(engineers):+.0f} ч</b>"
        )

            if shift == ShiftType.NIGHT:
                c.fill = night_fill; c.font = white_font
            elif shift == ShiftType.DAILY:
                c.fill = daily_fill; c.font = white_font
            elif shift == ShiftType.HOLIDAY:
                c.fill = holiday_fill; c.font = white_font
            else:
                c.fill = off_fill

    # Строка Д.Н.
    dow_row = len(engineers) + 2
    ws.cell(dow_row, 1, "Д.Н.").fill = name_fill
    ws.cell(dow_row, 1).font = bold_font
    ws.cell(dow_row, 1).alignment = center
    ws.cell(dow_row, 1).border = border

    dow_ru = ["Пн","Вт","Ср","Чт","Пт","Сб","Вс"]
    for d in range(1, days_in_month + 1):
        dt = date(year, month, d)
        c = ws.cell(dow_row, d + 1, dow_ru[dt.weekday()])
        c.alignment = center
        c.font = bold_font
        c.border = border

    # Ширина
    ws.column_dimensions["A"].width = 14
    for i in range(2, days_in_month + 2):
        ws.column_dimensions[openpyxl.utils.get_column_letter(i)].width = 4.5

    ws.freeze_panes = "B2"
    wb.save(path)

import requests

def load_from_isdayoff(year: int):
    url = f"https://isdayoff.ru/api/getdata?year={year}&cc=ru"
    response = requests.get(url)
    # Ответ: строка из 365 символов: 0 — рабочий, 1 — выходной, 2 — сокращённый
    data = {}
    d = date(year, 1, 1)
    for ch in response.text:
        data[d.isoformat()] = {
            "weekend": ch == "1",
            "holiday": ch == "1" and d.weekday() < 5,  # праздник в будень
            "short": ch == "2",
        }
        d += timedelta(days=1)
    return data
